  const STATUS_CFG = {
    confirmed:   { label:'Confirmed',   bg:'#E6F4EA', text:'#137333', border:'#CEEAD6', dot:'#0F9D58' },

  function StatusChip({ status }) {
    const cfg = STATUS_CFG[status] ?? { label:status, bg:'#F8F9FA', text:'#5F6368',
                                         border:'#E8EAED', dot:'#9AA0A6' }
    return (
      <Chip label={cfg.label} size="small" sx={{
        fontWeight:700, borderRadius:'8px', fontSize:'0.68rem', height:24,
        bgcolor: cfg.bg, color: cfg.text,
        border: `1px solid ${cfg.border}`,
        borderLeft: `3px solid ${cfg.dot}`,
      }} />
    )
  }

  const AVATAR_PALETTE = ['#1A73E8','#0F9D58','#9334E6','#FA7B17',
                          '#D93025','#009688','#F9AB00','#EA4335']

    AVATAR_PALETTE[(name.charCodeAt(0) || 0) % AVATAR_PALETTE.length]
  const nameInitials = (name = '') =>
    name.split(' ').slice(0,2).map(n => n[0] || '').join('').toUpperCase()

  <Avatar sx={{
    width, height,
    bgcolor: alpha(avatarBgColor(name), 0.15),
    color: avatarBgColor(name),
    fontWeight: 700, fontSize: size < 36 ? '0.68rem' : '0.85rem',
  }}>
    {nameInitials(name)}
  </Avatar>

  function ChartTooltip({ active, payload, label, labelFormatter }) {
    if (!active || !payload?.length) return null
    return (
      <Paper sx={{
        p:2, borderRadius:3, minWidth:160,
        border:'1px solid #E8EAED',
        boxShadow:'0 4px 20px rgba(32,33,36,0.18)',
        bgcolor:'#FFFFFF',
      }}>
        <Typography variant="caption" fontWeight={700}
          sx={{ color:'#202124', display:'block', mb:0.75 }}>
          {labelFormatter ? labelFormatter(label) : label}
        </Typography>
        {payload.map((entry) => (
          <Box key={entry.name} sx={{ display:'flex', alignItems:'center',
            gap:1, mb:0.25 }}>
            <Box sx={{ width:8, height:8, borderRadius:'50%',
              bgcolor:entry.color, flexShrink:0 }} />
            <Typography variant="caption" sx={{ color:'#5F6368' }}>
              {entry.name}:
            </Typography>
            <Typography variant="caption" fontWeight={700}
              sx={{ color:'#202124', ml:'auto' }}>
              {entry.value}
            </Typography>
          </Box>
        ))}
      </Paper>
    )
  }
